drone_motors.ino — Arduino UNO 
// ── Définition des broches ──────────────────────────
int M1 = 2;  // Moteur 1 — Avant-gauche
int M2 = 3;  // Moteur 2 — Avant-droite
int M3 = 4;  // Moteur 3 — Arrière-gauche
int M4 = 5;  // Moteur 4 — Arrière-droite

void setup() {
  pinMode(M1, OUTPUT);  // Broche 2 en sortie
  pinMode(M2, OUTPUT);  // Broche 3 en sortie
  pinMode(M3, OUTPUT);  // Broche 4 en sortie
  pinMode(M4, OUTPUT);  // Broche 5 en sortie
  Serial.begin(9600);
}

void loop() {
  // ── MONTER : tous les moteurs ON ──
  digitalWrite(M1, HIGH);
  digitalWrite(M2, HIGH);
  digitalWrite(M3, HIGH);
  digitalWrite(M4, HIGH);
  delay(3000);  // Vol pendant 3 secondes

  // ── ROTATION GAUCHE : M1+M3 OFF ──
  digitalWrite(M1, LOW);
  digitalWrite(M3, LOW);
  delay(3000);  // Rotation 3 secondes

  // ── ROTATION DROITE : M1+M3 ON, M2+M4 OFF ──
  digitalWrite(M1, HIGH);
  digitalWrite(M3, HIGH);
  digitalWrite(M2, LOW);
  digitalWrite(M4, LOW);
  delay(3000);  // Rotation 3 secondes

  // ── ATTERRISSAGE : tous les moteurs OFF ──
  digitalWrite(M1, LOW);
  digitalWrite(M2, LOW);
  digitalWrite(M3, LOW);
  digitalWrite(M4, LOW);
  delay(2000);
}  // Fin loop() → recommence depuis le début

══════════════════════════════════════════════════════════════════
int M1 = 2;   ► On attribue le numéro de broche à chaque moteur.
                 Cela permet d'utiliser M1 au lieu de "2" partout.
                 Plus lisible et plus facile à modifier.

 PARTIE 2 : setup() — S'exécute UNE SEULE FOIS au démarrage
══════════════════════════════════════════════════════════════════
pinMode(M1, OUTPUT);
► Configure la broche en mode SORTIE (OUTPUT).
  En mode sortie, la broche peut envoyer 5V (HIGH) ou 0V (LOW).
  Sans cette ligne, la broche est en mode entrée (ne marche pas).

══════════════════════════════════════════════════════════════════
 PARTIE 3 : loop() — Boucle infinie
══════════════════════════════════════════════════════════════════
digitalWrite(M1, HIGH);
► Envoie 5V sur la broche 2  →  M1 TOURNE

digitalWrite(M1, LOW);
► Envoie 0V sur la broche 2  →  M1 S'ARRÊTE

delay(3000);
► Attend 3000 millisecondes = 3 secondes avant la prochaine ligne.

Étape 2 — Ajouter la carte Arduino UNO

La carte Arduino est le cerveau du drone

1
Dans l'éditeur de circuit, regardez le panneau de droite : c'est la bibliothèque de composants.
2
Dans la barre de recherche, tapez "Arduino UNO" et appuyez sur Entrée.
3
Glissez-déposez la carte Arduino UNO depuis le panneau vers le centre du plan de travail.
4
Placez la carte au centre pour laisser de l'espace autour pour les composants.

Étape 3 — Ajouter les 4 moteurs DC

Un moteur DC par hélice du drone

1
Dans la bibliothèque, recherchez "DC Motor" ou "Motor".
2
Ajoutez 4 moteurs DC sur le plan de travail. Disposez-les en forme de croix autour de l'Arduino.
3
⚠️ Important — Tinkercad n'a pas de vrai moteur brushless. Utilisez soit un moteur DC soit une LED pour simuler chaque moteur.
4
Étiquetez chaque moteur : clic droit → "Annoter" → M1, M2, M3, M4.

Étape 4 — Câblage des connexions

Relier chaque moteur à sa broche Arduino

1
M1 (Avant-Gauche) : Cliquez sur la borne + du moteur M1, tirez un fil jusqu'à la Pin 2 de l'Arduino.
2
M2 (Avant-Droite) : Connectez la borne + du moteur M2 à la Pin 3.
3
M3 (Arrière-Gauche) : Connectez M3 à la Pin 4.
4
M4 (Arrière-Droite) : Connectez M4 à la Pin 5.
5
Connectez toutes les bornes GND (–) des moteurs au GND de l'Arduino.
⚡ Tableau des connexions
ComposantBorne→ArduinoCouleur fil
Moteur M1 (Av-G)+→ Pin 2 Cyan
Moteur M2 (Av-D)+→ Pin 3 Vert
Moteur M3 (Ar-G)+→ Pin 4 Orange
Moteur M4 (Ar-D)+→ Pin 5 Violet
Tous les moteurs–→ GND Noir
⚠️ Protection obligatoire : Dans un vrai circuit, ajoutez un transistor NPN (2N2222) ou un driver L298N entre chaque broche Arduino et le moteur, car les moteurs consomment plus de courant que la sortie Arduino ne peut fournir (max 40mA).

Étape 5 — Écrire le programme Arduino

Copiez ce code dans l'éditeur Tinkercad

1
Cliquez sur la carte Arduino dans Tinkercad, puis cliquez sur "Code" en haut de l'écran.
2
Sélectionnez le mode "Texte" (pas "Blocs") pour écrire du vrai code Arduino.
3
Effacez le code existant, puis copiez-collez le code ci-dessous.
4
Cliquez sur Démarrer la simulation pour lancer l'exécution.
Code à copier dans Tinkercad
drone_tinkercad.ino 
// ═══════════════════════════════════════════════
// SIMULATION DRONE ARDUINO — Tinkercad
// Moteurs DC — Broches 2, 3, 4, 5
// ═══════════════════════════════════════════════

// Déclaration des broches
const int M1 = 2;  // Moteur 1 — Avant-Gauche
const int M2 = 3;  // Moteur 2 — Avant-Droite
const int M3 = 4;  // Moteur 3 — Arrière-Gauche
const int M4 = 5;  // Moteur 4 — Arrière-Droite

// ── SETUP : s'exécute une seule fois ────────────
void setup() {
  Serial.begin(9600);
  pinMode(M1, OUTPUT);
  pinMode(M2, OUTPUT);
  pinMode(M3, OUTPUT);
  pinMode(M4, OUTPUT);
  Serial.println("=== Drone Arduino prêt ===");
}

// ── Fonction : allumer tous les moteurs ─────────
void monter() {
  digitalWrite(M1, HIGH);
  digitalWrite(M2, HIGH);
  digitalWrite(M3, HIGH);
  digitalWrite(M4, HIGH);
  Serial.println("ACTION: Monter — tous moteurs ON");
}

// ── Fonction : éteindre tous les moteurs ────────
void atterrir() {
  digitalWrite(M1, LOW);
  digitalWrite(M2, LOW);
  digitalWrite(M3, LOW);
  digitalWrite(M4, LOW);
  Serial.println("ACTION: Atterrir — tous moteurs OFF");
}

// ── Fonction : rotation gauche ───────────────────
void rotationGauche() {
  digitalWrite(M1, LOW);   // M1 OFF
  digitalWrite(M2, HIGH);  // M2 ON
  digitalWrite(M3, LOW);   // M3 OFF
  digitalWrite(M4, HIGH);  // M4 ON
  Serial.println("ACTION: Rotation Gauche");
}

// ── Fonction : rotation droite ───────────────────
void rotationDroite() {
  digitalWrite(M1, HIGH);  // M1 ON
  digitalWrite(M2, LOW);   // M2 OFF
  digitalWrite(M3, HIGH);  // M3 ON
  digitalWrite(M4, LOW);   // M4 OFF
  Serial.println("ACTION: Rotation Droite");
}

// ── LOOP : boucle infinie ────────────────────────
void loop() {
  monter();           // Décollage
  delay(3000);        // Maintien 3s

  rotationGauche();   // Tourner gauche
  delay(2000);

  monter();           // Remontée
  delay(1500);

  rotationDroite();   // Tourner droite
  delay(2000);

  atterrir();         // Atterrissage
  delay(2000);
}              // ← Retour au début de loop()
